cmake_minimum_required(VERSION 3.21)
project(HoloHand VERSION 0.1 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets Network DBus)
find_package(OpenCV 4.8 REQUIRED COMPONENTS core imgproc imgcodecs videoio video dnn)
find_package(X11 REQUIRED)
add_library(holohand-core src/core/Gesture.cpp)
target_include_directories(holohand-core PUBLIC src)
include(CTest)
add_executable(gesture-tests tests/gesture.cpp)
target_link_libraries(gesture-tests PRIVATE holohand-core)
target_compile_options(gesture-tests PRIVATE -UNDEBUG)
add_test(NAME gestures COMMAND gesture-tests)
add_executable(preview-tests tests/preview.cpp src/ui/HandPreview.cpp)
target_link_libraries(preview-tests PRIVATE holohand-core Qt6::Widgets)
target_compile_options(preview-tests PRIVATE -UNDEBUG)
add_test(NAME preview COMMAND preview-tests)
set_tests_properties(preview PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_library(holohand-vision src/vision/Tracker.cpp src/vision/Camera.cpp src/vision/MotionAligner.cpp)
target_include_directories(holohand-vision PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(holohand-vision PUBLIC holohand-core ${OpenCV_LIBS})
target_link_libraries(holohand-vision PUBLIC Qt6::Core)
option(HOLOHAND_OPENVINO "Accelerate tracking with the optional OpenVINO runtime" OFF)
if(HOLOHAND_OPENVINO)
 find_package(OpenVINO REQUIRED COMPONENTS Runtime)
 target_link_libraries(holohand-vision PRIVATE openvino::runtime)
 target_compile_definitions(holohand-vision PRIVATE HOLOHAND_OPENVINO=1)
endif()
add_executable(alignment-tests tests/alignment.cpp)
target_link_libraries(alignment-tests PRIVATE holohand-vision)
target_compile_options(alignment-tests PRIVATE -UNDEBUG)
add_test(NAME alignment COMMAND alignment-tests)
if(CMAKE_SYSTEM_NAME STREQUAL "FreeBSD")
 target_include_directories(holohand-vision PRIVATE /usr/local/include)
endif()
add_executable(holohand-benchmark src/vision/Benchmark.cpp)
target_link_libraries(holohand-benchmark PRIVATE holohand-vision)
if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
 set(INPUT_SOURCE src/input/LinuxInput.cpp src/input/LinuxInput.h)
else()
 set(INPUT_SOURCE src/input/X11Input.cpp src/input/InputGuardian.cpp)
endif()
add_executable(holohand src/main.cpp src/ui/App.cpp src/ui/App.h src/ui/HandPreview.cpp src/input/ManualOverride.cpp ${INPUT_SOURCE})
target_include_directories(holohand PRIVATE src ${X11_INCLUDE_DIR})
target_link_libraries(holohand PRIVATE holohand-vision Qt6::Widgets Qt6::Network Qt6::DBus ${X11_LIBRARIES} ${X11_XTest_LIB})
if(CMAKE_SYSTEM_NAME STREQUAL "FreeBSD")
 target_link_libraries(holohand PRIVATE ${X11_Xi_LIB})
endif()
install(TARGETS holohand holohand-benchmark RUNTIME DESTINATION bin)
install(DIRECTORY models/ DESTINATION share/holohand/models)
if(HOLOHAND_OPENVINO AND CMAKE_SYSTEM_NAME STREQUAL "Linux")
 # Keep the optional runtime private to this app; no system library changes.
 get_target_property(HOLOHAND_OV_LIBRARY openvino::runtime IMPORTED_LOCATION_RELEASE)
 get_filename_component(HOLOHAND_OV_LIBDIR "${HOLOHAND_OV_LIBRARY}" DIRECTORY)
 file(GLOB HOLOHAND_OV_LIBS
      "${HOLOHAND_OV_LIBDIR}/libopenvino.so.*"
      "${HOLOHAND_OV_LIBDIR}/libopenvino_intel_cpu_plugin.so"
      "${HOLOHAND_OV_LIBDIR}/libopenvino_onnx_frontend.so.*"
      "${HOLOHAND_OV_LIBDIR}/libtbb.so.*"
      "${HOLOHAND_OV_LIBDIR}/libtbbmalloc.so.*"
      "${HOLOHAND_OV_LIBDIR}/libtbbbind*.so.*"
      "${HOLOHAND_OV_LIBDIR}/libhwloc.so.*")
 install(FILES ${HOLOHAND_OV_LIBS} DESTINATION lib/holohand)
 set_target_properties(holohand holohand-benchmark PROPERTIES INSTALL_RPATH "$ORIGIN/../lib/holohand")
 install(DIRECTORY vendor/openvino/ DESTINATION share/holohand/licenses/openvino OPTIONAL)
endif()
